\begin{frame}{1 \textperiodcentered{} The economic problem}
A firm has a profitable project, but its bank will lend only up to a limit.
After exhausting that line, the firm faces a choice:

\medskip
\begin{center}
\Large
stop investing
\quad or \quad
pay more for another source of funds?
\end{center}

\medskip
The answer depends on one comparison:
\[
\text{marginal return at the bank limit}
\quad\text{versus}\quad
\text{cost of alternative finance}.
\]

\vfill
\centering
\alert{Access is not simply ``yes'' or ``no'': its cost determines the choice.}
\end{frame}

\begin{frame}{2 \textperiodcentered{} Question and answer}
\textbf{Question.} How does the cost of accessing alternative finance affect
investment and output for a firm facing a bank-credit limit?

\medskip
\textbf{Answer.} There is an access-cost threshold:
\begin{itemize}
  \item above it, the firm stops at the bank limit, so credit availability
        determines investment;
  \item below it, the firm invests beyond the limit, so the marginal financing
        cost determines investment.
\end{itemize}

\medskip
The model derives the threshold, the two optimal choices, and their comparative
statics.
\end{frame}

\begin{frame}{3 \textperiodcentered{} What motivates the model}
The thesis finds a preliminary qualitative pattern: credit volume and a credit
spread may contain different forecasting information across Peruvian sectors.

\medskip
\begin{block}{What this paper does}
It derives a mechanism linking financing access to investment responses and
states cross-sectional predictions that the thesis can test.
\end{block}

\begin{alertblock}{What this paper does not do}
It does not derive forecast accuracy, causal effects, or the horizon at which
an indicator should peak.
\end{alertblock}
\end{frame}

\begin{frame}{4 \textperiodcentered{} Agent, timing, and financing order}
At date $t$, a firm with internal funds $n$ chooses capital $k$, bank credit
$b$, and alternative finance $m$. Capital produces at $t+1$.
\[
\max_{k,b,m}\ \beta F(k)-\Rb b-(\Rb+\tau)m
\quad\text{s.t.}\quad
k=n+b+m,\quad 0\le b\le\bar b,\quad m\ge0.
\]
\[
 F'>0,\quad F''<0,\quad \tau\ge0.
\]
Bank finance is cheaper but capped. The wedge $\tau$ captures issuance,
information, collateral, and other costs of raising funds outside the bank.
\end{frame}

\begin{frame}{5 \textperiodcentered{} The access-cost threshold}
Let $\bar k=n+\bar b$ be capital after the bank line is exhausted. Focus on a
firm that still wants to invest at that point at the bank rate:
\[
\beta F'(\bar k)>\Rb.
\]

The next unit costs $\Rb+\tau$ outside the bank. It is profitable when
\[
\beta F'(\bar k)>\Rb+\tau.
\]

Therefore the threshold is
\[
\boxed{\tau^*=\beta F'(\bar k)-\Rb>0.}
\]

This single threshold replaces the earlier binary access/no-access cases.
\end{frame}

\begin{frame}{6 \textperiodcentered{} Main proposition}
\small
\begin{tabular}{p{0.27\linewidth}p{0.31\linewidth}p{0.31\linewidth}}
\toprule
 & Expensive access & Affordable access \\
\midrule
Condition & $\tau>\tau^*$ & $0\le\tau<\tau^*$ \\
Capital & $k^*=\bar k=n+\bar b$ & $\beta F'(k^*)=\Rb+\tau$ \\
Bank-line effect & $\partial k^*/\partial\bar b=1$ & $\partial k^*/\partial\bar b=0$ \\
Access-cost effect & $\partial k^*/\partial\tau=0$ locally &
 $\partial k^*/\partial\tau=1/[\beta F''(k^*)]<0$ \\
What determines scale? & bank-credit limit & marginal financing cost \\
\bottomrule
\end{tabular}

\medskip
At $\tau=\tau^*$ the policy is continuous but has a kink. Comparative statics
are local to a strict side of the threshold.
\end{frame}

\begin{frame}{7 \textperiodcentered{} Why the proposition is true}
After substituting the financing identity, profit has two slopes:
\[
\Pi'(k)=
\begin{cases}
\beta F'(k)-\Rb, & k<\bar k,\\
\beta F'(k)-(\Rb+\tau), & k>\bar k.
\end{cases}
\]

At the bank limit, the right-hand slope is
\[
\Pi'_+(\bar k)=\beta F'(\bar k)-\Rb-\tau=\tau^*-\tau.
\]

\begin{itemize}
  \item If $\tau\ge\tau^*$, the slope turns non-positive at the kink.
  \item If $\tau<\tau^*$, profit keeps increasing beyond the bank limit.
  \item Strict concavity makes the resulting optimum global and unique.
\end{itemize}

\vfill
\footnotesize\textbf{Proof status:} threshold and comparative statics complete;
formal sectoral aggregation and empirical discipline of $G_s$ remain open.
\end{frame}

\begin{frame}{8 \textperiodcentered{} Quadratic specialization}
Let
\[
F(k)=ak-\frac{c}{2}k^2,\qquad a,c>0.
\]
Then
\[
\tau^*=\beta\bigl[a-c(n+\bar b)\bigr]-\Rb,
\]
and the outside-finance target is
\[
k_M(\tau)=\frac{\beta a-\Rb-\tau}{\beta c}.
\]

Within the focal domain,
\[
\boxed{k^*(\tau)=\max\{n+\bar b,\,k_M(\tau)\}}.
\]
Equivalently,
\[
\boxed{k^*(\tau)-\bar k
=\max\left\{0,\frac{\tau^*-\tau}{\beta c}\right\}}.
\]

The distance below the threshold maps directly into additional investment and,
because $F'>0$, additional output.
\end{frame}

\begin{frame}{9 \textperiodcentered{} Numerical illustration}
For $\beta=.95$, $a=2$, $c=.5$, $n=.4$, $\bar b=.5$, and $\Rb=.5$:
\[
\bar k=.900,\qquad \tau^*=.9725.
\]

\begin{center}
\small
\begin{tabular}{lcc}
\toprule
 & Expensive access & Affordable access \\
\midrule
Access wedge & $\tau=1.10$ & $\tau=.90$ \\
Optimal capital & $k^*=.900$ & $k^*=1.053$ \\
Output & $F(k^*)=1.598$ & $F(k^*)=1.828$ \\
Larger bank line & raises capital locally & changes funding mix locally \\
Lower $\tau$ & no local scale effect & raises capital and output \\
\bottomrule
\end{tabular}
\end{center}

At $\tau=.90$, affordable access adds $0.153$ units of capital and $0.231$
units of output relative to stopping at the bank limit.

\medskip
The interesting object is the change in behavior at $\tau^*$, not the chosen
calibration.
\end{frame}

\begin{frame}{10 \textperiodcentered{} From firms to sectors}
Let firms in sector $s$ draw access wedges from $G_s(\tau)$. The constrained
share is generated by the threshold:
\[
\lambda_s
=\Pr(\tau_i\ge\tau^*)
=1-G_s(\tau^*).
\]

\begin{itemize}
  \item A distribution shifted toward high wedges produces more firms whose
        investment is governed by the bank limit.
  \item A distribution shifted toward low wedges produces more firms whose
        investment is governed by marginal financing costs.
  \item Tradable and non-tradable labels enter only here, as an empirical
        hypothesis about $G_T$ and $G_N$.
\end{itemize}

\alert{The model no longer assumes directly that an entire sector is one firm type.}
\end{frame}

\begin{frame}{11 \textperiodcentered{} Empirical implications}
The mechanism suggests sharper tests than comparing forecast peaks:
\begin{enumerate}
  \item Credit-quantity effects should be stronger among firms with exhausted
        bank lines and high alternative-finance wedges.
  \item Financing-cost effects should be stronger among firms already using
        bonds, foreign credit, or other non-bank sources.
  \item A larger bank line should raise total finance above $\tau^*$, but should
        mainly change debt composition below $\tau^*$.
\end{enumerate}

\medskip
The thesis's credit volume and spread are proxies for $\bar b$ and the relevant
financing wedge. The mapping must be tested rather than assumed.
\end{frame}

\begin{frame}{12 \textperiodcentered{} What the simulation checks}
\texttt{code/verify.py} is being aligned to verify:
\begin{itemize}
  \item the formula for $\tau^*$ and the identity $k_M(\tau^*)=\bar k$;
  \item the first-order condition on each side of the threshold;
  \item continuity and the change in local derivatives at the kink;
  \item every numerical value reported in the paper and slides.
\end{itemize}

\medskip
It cannot establish:
\begin{itemize}
  \item that $G_N$ is shifted above $G_T$ in Peru;
  \item that the observed spread equals $\tau$;
  \item causality or exact forecast horizons.
\end{itemize}
\end{frame}

\begin{frame}{13 \textperiodcentered{} Lean: statement and scope}
\small
\textbf{Paper statement:}
\[
\tau<\tau^*
\quad\Longleftrightarrow\quad
\bar k<k_M(\tau)
\qquad\text{(quadratic technology).}
\]

\textbf{Corresponding Lean declaration:}
\begin{block}{}
\ttfamily
theorem access\_below\_threshold\_iff\\
\quad (hbeta : 0 < beta) (hc : 0 < c) :\\
\quad tau < accessThreshold beta a c RB cap <->\\
\qquad cap < targetCapital beta a c (RB + tau)
\end{block}

Lean verifies the algebraic equivalence under positivity assumptions. It does
not verify the empirical interpretation of $\tau$ or the sectoral distribution.
\end{frame}

\begin{frame}{14 \textperiodcentered{} Where the AI claim was rejected}
\textbf{Initial AI claim:}
\[
\lambda_N>\lambda_T
\quad\text{was treated as the sectoral mechanism.}
\]

\textbf{Problem found by hand:} the ranking was assumed rather than generated.

\medskip
\textbf{Revision:}
\[
\lambda_s=1-G_s(\tau^*).
\]
The model now derives the constrained share from a distribution of access
costs. Whether $G_N$ differs from $G_T$ remains an empirical hypothesis.

\begin{frame}{15 \textperiodcentered{} Limitations and feedback sought}
\textbf{Known limits}
\begin{itemize}
  \item $\bar b$ and $\tau$ remain reduced-form primitives;
  \item the observed loan spread may not equal the alternative-finance wedge;
  \item the static model does not generate forecast horizons or causal effects.
\end{itemize}

\medskip
\textbf{Questions for the class}
\begin{enumerate}
  \item Is the observed spread a defensible proxy for $\tau$, or should the
        empirical mapping use another financing-cost measure?
  \item Which observable best disciplines $G_s$: firm size, exporter status,
        bond issuance, or foreign-credit access?
  \item Should the next extension endogenize $\bar b$ through net worth, or add
        investment dynamics first?
\end{enumerate}
\end{frame}
